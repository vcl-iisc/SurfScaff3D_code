A flat rectangular plate with two round through-holes (like a metal mounting bracket).

The hard part is the holes. A solid sheet cannot have a real hole, and splitting a disc into polar pie-slices gives a degenerate centre and a bulgy rim. So I cut each hole the clean way — a SQUARE-TO-CIRCLE RING: the plate's flat face around a hole is FOUR patches, each mapping one side of a square cell out to one quarter of the hole circle (no collapsed centre, no internal seam). Each hole's wall is a short tube connecting the top and bottom face rings, so the hole goes right through (a real see-through hole, not a recess).

**Coordinate system:**
- Z is up; the plate lies flat, bottom face at z=0 and top face at z=0.10 (the thickness).
- The plate spans x in [-1, 1], y in [-0.5, 0.5]; the two holes (radius 0.22) are centred at (-0.5, 0) and (0.5, 0).

**Construction:**
- Two square cells, x in [-1, 0] and x in [0, 1] (each 1x1), one hole per cell.
- For each cell, each flat face (top and bottom) is 4 ring patches (square side -> hole quarter-circle). Adjacent ring patches share their diagonal edges, and the two cells share the x=0 edge, so the faces tile the whole plate while leaving two circular gaps.
- Each hole wall is a tube whose top and bottom rings equal the top/bottom face inner arcs, so the hole is genuinely open through the plate.
- The plate's outer rim is side-wall strips joining the top and bottom faces around the perimeter.
- Smooth man-made surface, no fine texture, so s=0.97 and f=1 everywhere.

import numpy as np
patches = []
def emit(label, s, grid, f=1): patches.append((label, s, f, np.asarray(grid, float)))

H = 0.10                      # plate thickness (bottom z=0, top z=H)
R = 0.22                      # hole radius
A = 0.5                       # half-size of each (square) cell
NSEG, NRAD = 5, 4            # points along a square side / radial points (arc -> edge)
QUARTERS = [(-np.pi/4, np.pi/4), (np.pi/4, 3*np.pi/4),
            (3*np.pi/4, 5*np.pi/4), (5*np.pi/4, 7*np.pi/4)]
CORNERS = {"xpos": ((1,-1),(1,1)), "ypos": ((1,1),(-1,1)),
           "xneg": ((-1,1),(-1,-1)), "yneg": ((-1,-1),(1,-1))}
SIDES = ["xpos", "ypos", "xneg", "yneg"]

# one flat face of a cell = 4 patches, each blending the hole's quarter-arc out to a square side
def ring_face(cx, cy, z, nm):
    for k, side in enumerate(SIDES):
        (c0x,c0y),(c1x,c1y) = CORNERS[side]; t0,t1 = QUARTERS[k]; g = []
        for i in range(NRAD):
            u = i/(NRAD-1); row = []
            for j in range(NSEG):
                v = j/(NSEG-1); th = t0 + v*(t1-t0)
                ax_, ay_ = cx + R*np.cos(th), cy + R*np.sin(th)                 # inner: hole arc
                ex_, ey_ = cx + (c0x+v*(c1x-c0x))*A, cy + (c0y+v*(c1y-c0y))*A   # outer: square side
                row.append(((1-u)*ax_+u*ex_, (1-u)*ay_+u*ey_, z))
            g.append(row)
        emit(f"plate/face/{nm}/{side}_{'top' if z>0 else 'bot'}", 0.97, g)

def hole_circle(cx, cy):                 # the hole rim, walking all four quarters (no duplicates)
    pts = []
    for qi,(t0,t1) in enumerate(QUARTERS):
        for j in range(NSEG):
            if qi>0 and j==0: continue
            th = t0 + j/(NSEG-1)*(t1-t0); pts.append((cx+R*np.cos(th), cy+R*np.sin(th)))
    return pts

def tube(cx, cy, nm):                    # hole wall: rims at z=0 and z=H equal the face inner arcs
    circ = hole_circle(cx, cy)           # already closed (last point 315deg == first -45deg)
    emit(f"plate/hole/{nm}/wall", 0.97, [[(x,y,0.0) for x,y in circ], [(x,y,H) for x,y in circ]])

def wall(side, cx):                      # outer side-wall strip along one square side of a cell
    (c0x,c0y),(c1x,c1y) = CORNERS[side]
    edge = [(cx+(c0x+v/(NSEG-1)*(c1x-c0x))*A, (c0y+v/(NSEG-1)*(c1y-c0y))*A) for v in range(NSEG)]
    emit(f"plate/edge/{side}_{int(cx*10)}", 0.97, [[(x,y,H) for x,y in edge], [(x,y,0.0) for x,y in edge]])

for nm, cx in [("left", -0.5), ("right", 0.5)]:
    ring_face(cx, 0.0, H, nm)            # top face (4 ring patches)
    ring_face(cx, 0.0, 0.0, nm)          # bottom face (4 ring patches)
    tube(cx, 0.0, nm)                    # hole wall
# outer perimeter walls (the x=0 edge is internal/shared, so it gets no wall)
wall("xneg", -0.5); wall("ypos", -0.5); wall("yneg", -0.5)
wall("xpos",  0.5); wall("ypos",  0.5); wall("yneg",  0.5)

with open("final_output.txt", "w") as f:
    f.write(f"<faces>\n{len(patches)}\n")
    for label, s, ff, g in patches:
        g = np.asarray(g, float)
        f.write(f"# {label}  s={s:.2f} f={ff}\n{g.shape[0]} {g.shape[1]}\n")
        for x, y, z in g.reshape(-1, 3):
            f.write(f"{x:.3f} {y:.3f} {z:.3f}\n")
    f.write("</faces>\n")
